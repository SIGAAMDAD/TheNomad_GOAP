cmake_minimum_required(VERSION 3.23)
cmake_policy(SET CMP0091 NEW)
option(NMDGOAP_MSVC_STATIC_CRT "Use static MSVC CRT" ON)
if(NMDGOAP_MSVC_STATIC_CRT)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
project(TheNomad_GOAP VERSION 0.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(NMDGOAP_BUILD_SMOKE "Build EASTL/EABase smoke fixture" ON)
# EABase uses cmake_minimum_required(VERSION 3.1). CMake 4 removed
# compatibility below 3.5; retain the immutable upstream source.
if(CMAKE_VERSION VERSION_GREATER_EQUAL "4.0")
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Legacy EABase compatibility")
endif()
include(cmake/Dependencies.cmake)
add_library(nmdgoap_build_contract INTERFACE)
target_compile_features(nmdgoap_build_contract INTERFACE cxx_std_17)
target_link_libraries(nmdgoap_build_contract INTERFACE EASTL EABase)
enable_testing()
if(NMDGOAP_BUILD_SMOKE)
  # G01 link/load proof; exported symbol is a test probe, NOT the G02 ABI.
  add_library(nmdgoap_bootstrap_core STATIC tests/bootstrap_core.cpp)
  target_link_libraries(nmdgoap_bootstrap_core PRIVATE nmdgoap_build_contract)
  add_library(nmdgoap_bootstrap_shared SHARED tests/bootstrap_shared.cpp)
  target_link_libraries(nmdgoap_bootstrap_shared PRIVATE nmdgoap_bootstrap_core nmdgoap_build_contract)
  add_executable(nmdgoap_dynamic_smoke tests/dynamic_smoke.cpp)
  if(CMAKE_DL_LIBS)
    target_link_libraries(nmdgoap_dynamic_smoke PRIVATE ${CMAKE_DL_LIBS})
  endif()
  add_test(NAME nmdgoap_dynamic_smoke
           COMMAND nmdgoap_dynamic_smoke "$<TARGET_FILE:nmdgoap_bootstrap_shared>")
  add_executable(nmdgoap_dependency_smoke tests/dependency_smoke.cpp)
  target_link_libraries(nmdgoap_dependency_smoke PRIVATE nmdgoap_build_contract)
  if(MSVC)
    target_compile_options(nmdgoap_dependency_smoke PRIVATE /W4 /permissive-)
  else()
    target_compile_options(nmdgoap_dependency_smoke PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME nmdgoap_dependency_smoke COMMAND nmdgoap_dependency_smoke)
endif()
